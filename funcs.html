// Variadic function
func sum(nums ...int) int {
    total := 0
    for _, n := range nums { total += n }
    return total
}
sum(1, 2, 3)        // = 6
nums := []int{1, 2, 3}
sum(nums...)        // unpack slice

// Function type
type Predicate func(int) bool

func filter(nums []int, fn Predicate) []int {
    result := []int{}
    for _, n := range nums {
        if fn(n) { result = append(result, n) }
    }
    return result
}

isEven := func(n int) bool { return n%2 == 0 }
evens := filter([]int{1,2,3,4}, isEven) // [2,4]

Hàm lồng nhau, biến capture, counter, option pattern

// Closure capture by reference
func makeCounter() func() int {
    count := 0
    return func() int {
        count++
        return count
    }
}
c := makeCounter()
c() // 1
c() // 2

// Loop variable tự động rebind mỗi iteration
callbacks := []func(){}
for i := 0; i < 3; i++ {
    callbacks = append(callbacks, func() { fmt.Println(i) })
}
for _, cb := range callbacks { cb() } // 0, 1, 2

// Functional Options Pattern
type Option func(*Server)
func WithPort(p int) Option {
    return func(s *Server) { s.port = p }
}
s := NewServer(WithPort(8080))

// Multiple return values
func divide(a, b float64) (float64, error) {
    if b == 0 { return 0, errors.New("division by zero") }
    return a / b, nil
}

// Named return — tự động khai báo biến
func minMax(nums []int) (min, max int) {
    min, max = nums[0], nums[0]
    for _, n := range nums[1:] {
        if n < min { min = n }
        if n > max { max = n }
    }
    return  // naked return: trả về min, max
}
Method expressions & values
Value receiver vs pointer receiver, method set
▼
receivermethod setvalue/pointertrung cấp
type Point struct { X, Y float64 }

// Value receiver — không thay đổi gốc
func (p Point) Distance() float64 {
    return math.Sqrt(p.X*p.X + p.Y*p.Y)
}

// Pointer receiver — có thể thay đổi
func (p *Point) Scale(factor float64) {
    p.X *= factor; p.Y *= factor
}

// Method expression (function với receiver là arg đầu)
dist := Point.Distance
dist(Point{3, 4})    // = 5.0

// Method value (bound to specific instance)
p := &Point{3, 4}
scale := p.Scale
scale(2)              // p.X=6, p.Y=8
Quy tắc: Nếu bất kỳ method nào cần pointer receiver, nên dùng pointer receiver cho toàn bộ type đó. Pointer receiver có method set lớn hơn value receiver.
