type Animal struct { Name string }
func (a Animal) Speak() string { return a.Name + " speaks" }

type Dog struct {
    Animal           // anonymous field (embedded)
    Breed string
}

d := Dog{Animal: Animal{"Rex"}, Breed: "Lab"}
d.Speak()     // promoted: d.Animal.Speak()
d.Name        // promoted field: d.Animal.Name

Interface & implicit implementation
Duck typing, nil interface, interface composition
▼
interfaceduck typingnil interfacetrung cấp
type Stringer interface {
    String() string
}

type Person struct { Name, Email string }
func (p Person) String() string {
    return p.Name + " <" + p.Email + ">"
}
// Person tự động implement Stringer (không cần khai báo)

// Nil interface vs nil pointer trong interface
var s Stringer           // nil: (type=nil, value=nil)
var p *Person            // nil pointer
var s2 Stringer = p     // NOT nil! (type=*Person, value=nil)
s2 != nil               // true — BẪY PHỔ BIẾN!

// Interface comparison
var a, b Stringer
a == b // so sánh cả type và value
Bẫy: Một interface chứa non-nil pointer sẽ KHÔNG bằng nil dù pointer đó là nil. Luôn kiểm tra error bằng err != nil chứ không phải err == nil sau khi assign.

// Generic function
func Map[T, U any](s []T, fn func(T) U) []U {
    result := make([]U, len(s))
    for i, v := range s { result[i] = fn(v) }
    return result
}

// Type inference
nums := Map([]int{1,2,3}, func(n int) string {
    return fmt.Sprint(n)
})

// Generic struct
type Stack[T any] struct { items []T }
func (s *Stack[T]) Push(v T) { s.items = append(s.items, v) }
func (s *Stack[T]) Pop() (T, bool) {
    if len(s.items) == 0 { var zero T; return zero, false }
    n := len(s.items) - 1
    v := s.items[n]
    s.items = s.items[:n]
    return v, true
}

var s Stack[int]
s.Push(1); s.Push(2)
v, _ := s.Pop() // v = 2
