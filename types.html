Type switch
switch v.(type) — xử lý nhiều loại interface
▼
type switchinterfacetrung cấp
func describe(i any) string {
    switch v := i.(type) {
    case int:
        return fmt.Sprintf("int: %d", v)
    case string:
        return fmt.Sprintf("string: %q", v)
    case []byte:
        return fmt.Sprintf("bytes: %d", len(v))
    case nil:
        return "nil"
    default:
        return fmt.Sprintf("unknown: %T", v)
    }
}

// any = interface{}
func Print(v any) { fmt.Println(v) }

// comparable — có thể dùng == và !=
// slice, map, func KHÔNG comparable
func Contains[T comparable](slice []T, item T) bool {
    for _, v := range slice {
        if v == item { return true }
    }
    return false
}

// Union constraint
type Number interface {
    int | int32 | int64 | float32 | float64
}

// Underlying type với ~

type Integer interface { ~int }

Channel types
chan, chan<-, <-chan, buffered/unbuffered
▼
chanbuffereddirectionaltrung cấp
// Unbuffered: sync communication
ch := make(chan int)

// Buffered: async up to capacity
bch := make(chan string, 10)

// Directional channels
var send chan<- int = ch  // send-only
var recv <-chan int = ch  // receive-only

// Range over channel
go func() {
    for v := range ch { // block cho đến khi close
        fmt.Println(v)
    }
}()

// Check if channel has value
select {
case v, ok := <-ch:
    if !ok { /* channel closed */ }
default: // non-blocking
}
