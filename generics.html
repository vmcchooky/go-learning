// Đa type parameter
func Reduce[T, R any](
    slice []T,
    initial R,
    fn func(R, T) R,
) R {
    result := initial
    for _, v := range slice {
        result = fn(result, v)
    }
    return result
}

total := Reduce([]int{1,2,3}, 0,
    func(acc, n int) int { return acc + n })

// Custom constraint với method

    int | float64 | string
}

func Min[T Ordered](a, b T) T {
    if a < b { return a }
    return b
}

Iter package & range-over-func
iter.Seq, iter.Seq2, custom iterators với for-range
▼
iter.Seqrange-over-funcnâng cao
import "iter"

// iter.Seq[V] = func(yield func(V) bool)
func Fibonacci() iter.Seq[int] {
    return func(yield func(int) bool) {
        a, b := 0, 1
        for {
            if !yield(a) { return }
            a, b = b, a+b
        }
    }
}

// Range over custom iterator
for n := range Fibonacci() {
    if n > 100 { break }
    fmt.Println(n)
}

// iter.Seq2[K, V] cho key-value pairs
func Enumerate[T any](s []T) iter.Seq2[int, T] {
    return func(yield func(int, T) bool) {
        for i, v := range s {
            if !yield(i, v) { return }
        }
    }
}

for i, v := range Enumerate([]string{"a","b"}) {
    fmt.Println(i, v)
}

// slices.Collect — tạo slice từ iterator
all := slices.Collect(Fibonacci())
Range-over-func là tính năng mạnh mẽ cho custom iterators, thay thế pattern dùng channel-based iteration hiệu quả hơn nhiều.

type Node[T any] struct {
    Value T
    Next  *Node[T]
}

// Type alias đơn giản (không generic) — đã có từ 1.9
type IntNode = Node[int]

// Generic type alias
type GenericNode[T any] = Node[T]

// Dùng generic type alias
var n1 GenericNode[string] = Node[string]{Value: "hello"}
var n2 IntNode = Node[int]{Value: 42}

// Useful pattern: alias cho generic container
type Result[T any] struct {
    Data  T
    Error error
}
type IntResult = Result[int]  // cụ thể hóa

func fetchInt() IntResult {
    return IntResult{Data: 42, Error: nil}
}

import (
    "slices"
    "maps"
    "cmp"
)

// slices package
s := []int{3, 1, 4, 1, 5}
slices.Sort(s)
slices.SortFunc(s, func(a, b int) int { return a - b })
idx, found := slices.BinarySearch(s, 3)
slices.Contains(s, 4)
s2 := slices.Clone(s)
s = slices.Compact(s) // remove consecutive duplicates, phải gán lại

// maps package
m := map[string]int{"a": 1, "b": 2}
keys := slices.Collect(maps.Keys(m))   // []string, thứ tự không đảm bảo
vals := slices.Collect(maps.Values(m)) // []int, thứ tự không đảm bảo
m2 := maps.Clone(m)
maps.DeleteFunc(m, func(k string, v int) bool {
    return v == 0
})

// cmp package
cmp.Compare(1, 2)   // -1, 0, 1
cmp.Min(3, 5)       // 3
cmp.Max(3, 5)       // 5

// min/max built-in
x := min(1, 2, 3)  // 1 (built-in, multi-arg)
y := max(1, 2, 3)  // 3
