Reflection (reflect)
reflect.Type, reflect.Value, struct field inspection
▼
reflectTypeOfValueOfnâng cao
import "reflect"

func inspect(v any) {
    t := reflect.TypeOf(v)
    val := reflect.ValueOf(v)

    fmt.Println(t.Kind())       // struct, int, slice...
    fmt.Println(t.Name())       // type name

    if t.Kind() == reflect.Struct {
        for i := 0; i < t.NumField(); i++ {
            f := t.Field(i)
            fmt.Printf("%s %s = %v tag=%s\n",
                f.Name, f.Type,
                val.Field(i).Interface(),
                f.Tag.Get("json"))
        }
    }
}

// Set value via reflection
func setField(obj any, name string, val any) {
    v := reflect.ValueOf(obj).Elem()  // deref pointer
    f := v.FieldByName(name)
    if f.CanSet() {
        f.Set(reflect.ValueOf(val))
    }
}

import "unsafe"

// String ↔ []byte zero-copy (read-only!)
func StringToBytes(s string) []byte {
    return unsafe.Slice(unsafe.StringData(s), len(s))
}

// Struct field offset
type T struct { A, B, C int }
var t T
offsetB := unsafe.Offsetof(t.B)  // bytes từ đầu struct

// Kích thước
unsafe.Sizeof(t)       // kích thước struct
unsafe.Alignof(t.B)   // alignment requirement

// Cảnh báo: unsafe phá vỡ type safety
// Chỉ dùng khi cần thiết cho performance
Cảnh báo: unsafe có thể gây crash, race condition, undefined behavior. Chỉ dùng khi đã benchmark và không có cách khác.

Testing & table-driven tests
testing.T, table-driven, benchmarks, fuzzing, testify
▼
testingbenchmarktable-driventrung cấp
func TestAdd(t *testing.T) {
    tests := []struct {
        name    string
        a, b    int
        want    int
    }{
        {"positive", 1, 2, 3},
        {"negative", -1, -2, -3},
        {"zero", 0, 0, 0},
    }
    for _, tc := range tests {
        t.Run(tc.name, func(t *testing.T) {
            got := Add(tc.a, tc.b)
            if got != tc.want {
                t.Errorf("Add(%d,%d)=%d, want %d",
                    tc.a, tc.b, got, tc.want)
            }
        })
    }
}

// Benchmark
func BenchmarkAdd(b *testing.B) {
    for i := 0; i < b.N; i++ {
        Add(1, 2)
    }
}

// Default logger
slog.Info("user logged in", "user_id", 123, "ip", "127.0.0.1")
slog.Error("db error", "err", err)

// JSON handler
logger := slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{
    Level: slog.LevelDebug,
}))
logger.Info("request",
    "method", "GET",
    "path", "/api/v1/users",
    "duration_ms", 42,
)

// Group attributes
logger.Info("request", slog.Group("http",
    "method", "POST",
    "status", 200,
))
HTTP Routing (ServeMux)
Method-based routing, path parameters, wildcard patterns
▼
httproutingServeMuxtrung cấp
mux := http.NewServeMux()

// Method + path pattern
mux.HandleFunc("GET /users", listUsers)
mux.HandleFunc("POST /users", createUser)
mux.HandleFunc("GET /users/{id}", getUser)
mux.HandleFunc("PUT /users/{id}", updateUser)
mux.HandleFunc("DELETE /users/{id}", deleteUser)

// Wildcard {id} — truy cập bằng PathValue
func getUser(w http.ResponseWriter, r *http.Request) {
    id := r.PathValue("id")
    fmt.Fprintf(w, "User ID: %s", id)
}

// Wildcard {path...} — match phần còn lại
mux.HandleFunc("GET /static/{path...}", serveStatic)

func serveStatic(w http.ResponseWriter, r *http.Request) {
    path := r.PathValue("path")
    http.ServeFile(w, r, "static/"+path)
}

http.ListenAndServe(":8080", mux)
ServeMux mới có thể thay thế nhiều trường hợp dùng gorilla/mux hay chi mà không cần dependency bên ngoài.

Profile-Guided Optimization (PGO)
Tối ưu dựa trên profile thực tế, cải thiện 2–14% performance
▼
pgooptimizationprofilenâng cao
// Bước 1: Thu thập CPU profile từ workload thật
// curl http://localhost:6060/debug/pprof/profile?seconds=30 > default.pgo

// Bước 2: Build với PGO
// go build -pgo=default.pgo .

// Hoặc dùng pprof tự động từ benchmark:
// go test -bench=. -cpuprofile=cpu.pgo ./...

// PGO không yêu cầu đổi source code

PGO là cách hiệu quả nhất để cải thiện performance mà không cần rewrite code. Chỉ cần build lại với profile từ production.
