// Xem escape analysis
// go build -gcflags="-m" ./...

// Object thường lên heap khi compiler thấy value "escape":
// 1. Trả về pointer ra ngoài function
func newFoo() *Foo {
    f := Foo{}   // thường escape vì địa chỉ thoát khỏi function
    return &f
}

// 2. Boxing vào interface có thể làm value escape
var i interface{} = Foo{}  // tùy ngữ cảnh, compiler quyết định

// 3. Backing array mà compiler không chứng minh được nằm trên stack
s := make([]byte, n)       // n không constant thường khiến backing array lên heap

// Tối ưu: dùng stack nếu có thể
func sum(nums []int) int {
    // Trả về value, không phải pointer
    total := 0           // thường ở stack nếu không escape
    for _, n := range nums { total += n }
    return total
}
sync.Pool
Object pooling, GC interaction, buffer reuse
▼
sync.PoolpoolingGCnâng cao
var bufPool = sync.Pool{
    New: func() any {
        return &bytes.Buffer{}
    },
}

func encode(data any) ([]byte, error) {
    buf := bufPool.Get().(*bytes.Buffer)
    defer func() {
        buf.Reset()
        bufPool.Put(buf)
    }()

    if err := json.NewEncoder(buf).Encode(data); err != nil {
        return nil, err
    }
    result := make([]byte, buf.Len())
    copy(result, buf.Bytes())
    return result, nil
}

// Lưu ý: Pool có thể bị GC bỏ bớt giữa các cycle
// Phù hợp cho: buffers, temporary objects
// Không dùng như cache bền vững, connection, state quan trọng

import (
    "net/http"
    _ "net/http/pprof"  // đăng ký endpoints
    "runtime"
    "runtime/pprof"
)

// HTTP pprof server
go http.ListenAndServe("localhost:6060", nil)

// CLI profiling
// go tool pprof http://localhost:6060/debug/pprof/heap
// go tool pprof http://localhost:6060/debug/pprof/goroutine
// go tool pprof -http=:8080 cpu.prof

// CPU profile trong code
f, _ := os.Create("cpu.prof")
pprof.StartCPUProfile(f)
defer pprof.StopCPUProfile()

// Memory profile
runtime.GC()
f2, _ := os.Create("mem.prof")
pprof.WriteHeapProfile(f2)
